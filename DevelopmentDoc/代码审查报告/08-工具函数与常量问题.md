**问题分析**:
- 定义了错误类但未使用，代码重复
- 没有利用自定义错误类的优势
- 错误处理不一致

**修改建议**:
在 Server Actions 中使用自定义错误类

**修改思路**:
```typescript
import { AppError, ValidationError, NotFoundError } from '@/lib/errors';

export async function updateLink(
  id: string,
  data: UpdateLinkInput
): Promise<ActionResult<Link>> {
  try {
    const validated = updateLinkSchema.parse(data);
    const session = await auth();

      throw new AppError("Authentication required", "AUTH_ERROR", 401);
    }

    const existingLink = await prisma.link.findUnique({
      where: { id },
      select: { userId: true },
    });

    if (!existingLink) {
      throw new NotFoundError("Link");
    }

    if (existingLink.userId !== session.user.id) {
      throw new AppError("Not authorized", "FORBIDDEN", 403);
    }

    // ... 更新逻辑

  } catch (error) {
    // 统一错误处理

      return { success: false, error: error.message };
    }
    if (error instanceof z.ZodError) {
      return { success: false, error: error.issues[0]?.message ?? "Validation failed" };
    }
    console.error("updateLink error:", error);
    return { success: false, error: "Failed to update link" };
  }
```

### 3. lib/utils.ts 中的 cn 函数依赖可能过大

**文件**: `src/lib/utils.ts`
**行号**: 1-14

**问题描述**:
`cn` 函数依赖 `clsx` 和 `tailwind-merge`，但整个包的体积较大。

**当前代码**:
```typescript
import { clsx, type ClassValue } from "clsx";
import { twMerge } from "tailwind-merge";

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}
```

**问题分析**:
- `clsx` 和 `tailwind-merge` 是生产依赖
- 可以用更轻量的实现替代

**修改建议**:
考虑使用原生实现，或确认依赖的必要性

**修改思路**:
```typescript
// 方案1: 使用更轻量的实现
export function cn(...classes: (string | undefined | null | false)[]) {
  return classes
    .filter(Boolean)
    .join(' ')
    .replace(/(\w+)-(\w+)/g, '$1$2')  // 处理 Tailwind 冲突
    .split(' ')
    .filter((cls, i, arr) => arr.indexOf(cls) === i);  // 去重
}

// 方案2: 确认依赖必要性（通常保持现状更好）
// tailwind-merge 对处理 Tailwind 类名冲突很有用
```

### 4. 缺少日期格式化工具

**文件**: `src/lib/`

**问题描述**:
项目中没有统一的日期格式化工具，各处使用不同的格式。

**问题分析**:
- 代码中直接使用 `new Date()` 或 `toLocaleString()`
- 日期格式不统一
- 时区处理不一致

**修改建议**:
添加日期格式化工具

**修改思路**:
```typescript
// src/lib/dates.ts

// 日期格式配置
export const DATE_FORMATS = {
  FULL: 'YYYY-MM-DD HH:mm:ss',
  DATE: 'YYYY-MM-DD',
  TIME: 'HH:mm:ss',
  RELATIVE: 'relative',
} as const;

/**
 * 格式化日期
 */
export function formatDate(
  date: Date | string | number,
  format: keyof typeof DATE_FORMATS = 'FULL'
): string {
  const d = typeof date === 'object' ? date : new Date(date);

  switch (format) {
    case 'DATE':
      return d.toISOString().split('T')[0];
    case 'TIME':
      return d.toTimeString().split(' ')[0];
    case 'RELATIVE':
      return formatRelative(d);
    default:
      return d.toISOString().replace('T', ' ').split('.')[0];
  }
}

/**
 * 格式化相对时间（如"2小时前"）
 */
export function formatRelative(date: Date): string {
  const now = new Date();
  const diff = now.getTime() - date.getTime();
  const seconds = Math.floor(diff / 1000);
  const minutes = Math.floor(seconds / 60);
  const hours = Math.floor(minutes / 60);
  const days = Math.floor(hours / 24);

  if (days > 7) {
    return formatDate(date, 'DATE');
  }
  if (days > 0) return `${days}天前`;
  if (hours > 0) return `${hours}小时前`;
  if (minutes > 0) return `${minutes}分钟前`;
  return '刚刚';
}
```

```typescript
// src/lib/url.ts

/**
 * 确保 URL 有协议前缀
 */
export function ensureProtocol(url: string, protocol: 'http' | 'https' = 'https'): string {
  if (!url) return '';
  if (/^https?:\/\//i.test(url)) return url;
  return `${protocol}:${url.startsWith('//') ? '' : '//'}${url}`;
}

/**
 * 获取域名的显示名称
 */
export function getDomainName(url: string): string {
  try {
    const urlObj = new URL(url);
    return urlObj.hostname.replace('www.', '');
  } catch {
    return url;
  }
}

/**
 * 验证 URL 是否为图片
 */
export function isImageUrl(url: string): boolean {
  const imageExtensions = ['.jpg', '.jpeg', '.png', '.gif', '.webp', '.svg'];
  const lowerUrl = url.toLowerCase();
  return imageExtensions.some(ext => lowerUrl.endsWith(ext)) ||
    lowerUrl.includes('/image/') ||
    lowerUrl.includes('/img/');
}
```
