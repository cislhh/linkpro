#### 2. ⚠️ 头像上传逻辑复杂

**问题** (第 98-161 行):
```typescript
const handleAvatarUpload = async (file: File) => {
  // 验证文件大小
  if (file.size > AVATAR_SPEC.maxSize) {
    toast.error("文件过大", {
      description: `最大文件大小为 ${AVATAR_SPEC.maxSize / 1024 / 1024}MB`,
    });
    return;
  }

  // 验证文件类型
  const isValidType = AVATAR_SPEC.allowedFormats.some(
    (format) => format === file.type
  );
  if (!isValidType) {
    toast.error("不支持的文件格式", {
      description: "支持的格式: JPG, PNG, WebP",
    });
    return;
  }

  setIsUploading(true);

  try {
    const formData = new FormData();
    formData.append("file", file);

    const result = await uploadAvatar(formData);

    if (result.success) {
      // Auto-save avatar URL to database
      const currentValues = form.getValues();
      const updateResult = await updateUserProfile({
        ...currentValues,
        avatarUrl: result.data.url,
      } as UpdateProfileInput);

      if (updateResult.success) {
        // 更新 store
        const { name, bio, avatarUrl, phone, contact } = updateResult.data;
        setUserProfile({ name, bio, avatarUrl, phone, contact });

        // Update form with returned data
        form.reset({
          name: updateResult.data.name || "",
          bio: updateResult.data.bio || "",
          avatarUrl: updateResult.data.avatarUrl || "",
          phone: updateResult.data.phone || "",
          contact: updateResult.data.contact || "",
        });
        setAvatarPreview(updateResult.data.avatarUrl);
        toast.success("头像上传成功");
      } else {
        toast.error("保存失败", { description: updateResult.error });
      }
    } else {
      toast.error("上传失败", { description: result.error });
    }
  } catch (error) {
    console.error("Avatar upload error:", error);
    toast.error("上传失败", { description: "请稍后重试" });
  } finally {
    setIsUploading(false);
  }
};
```

**问题分析**:
1. **混合职责**: 验证、上传、保存都在一起
2. **自动保存**: 头像上传后自动保存整个表单，可能不符合用户预期
3. **重复逻辑**: 更新 store 和 form 的代码重复

**建议改进**:
```typescript
// hooks/use-avatar-upload.ts
export function useAvatarUpload() {
  const [isUploading, setIsUploading] = useState(false);
  const [avatarUrl, setAvatarUrl] = useState<string | null>(null);

  const uploadAvatar = async (file: File) => {
    // 验证逻辑
    validateAvatarFile(file);

    setIsUploading(true);
    try {
      const result = await uploadAvatarAction(file);
      if (result.success) {
        setAvatarUrl(result.data.url);
        return result.data.url;
      }
      throw new Error(result.error);
    } finally {
      setIsUploading(false);
    }
  };

  return { uploadAvatar, isUploading, avatarUrl };
}
```

#### 3. ⚠️ Store 使用不完整

**问题** (第 72-73 行):
```typescript
const profile = useUserStore((state) => state.profile);
```

**问题分析**:

2. 表单组件直接从 store 读取

## 总体重构建议

### 1. 组件拆分方案

```
src/components/features/profile/
├── profile-form.tsx (主入口，~150 行)
├── avatar-upload-section.tsx (~100 行)
├── profile-fields.tsx (~100 行)
└── hooks/
    ├── use-profile-form.ts (~80 行)
    └── use-avatar-upload.ts (~60 行)
```

### 2. 头像上传流程优化

**当前流程**:
```
用户选择文件 → 上传 → 自动保存整个表单 → 更新 store
```

**建议流程**:
```
用户选择文件 → 上传 → 更新表单字段 (仅预览)
                                     ↓
                          用户点击保存 → 保存整个表单
```
