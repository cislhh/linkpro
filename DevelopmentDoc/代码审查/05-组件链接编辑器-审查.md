```typescript
// ✅ 方案 1: 使用乐观更新
const onSubmit = async (data: LinkFormValues) => {
  const tempId = `temp-${Date.now()}`;
  const optimisticLink: Link = {
    id: tempId,
    ...data,
    createdAt: new Date(),
    updatedAt: new Date(),
  };

  // 乐观更新
  setLinks((prev) => isEditing ? prev : [...prev, optimisticLink]);

  try {
    if (isEditing && editingLink) {
      const result = await updateLink(editingLink.id, data);
      if (!result.success) throw new Error(result.error);

      // 用服务器数据替换
      setLinks((prev) =>
        prev.map((link) =>
          link.id === tempId ? result.data : link
        )
      );
    } else {
      const result = await createLink(data);
      if (!result.success) throw new Error(result.error);

      // 用服务器数据替换
      setLinks((prev) =>
        prev.map((link) =>

    toast.error(error.message);
  }
};
```

## 总体建议

### 1. 统一链接数据管理

**当前问题**:
- 多个组件都调用 `getUserLinks()`
- 每次更新后都重新获取
- 没有统一的状态管理

**建议架构**:
```typescript
// src/contexts/links-context.tsx
interface LinksContextValue {
  links: Link[];
  isLoading: boolean;
  error: string | null;
  addLink: (data: CreateLinkInput) => Promise<void>;
  updateLink: (id: string, data: UpdateLinkInput) => Promise<void>;
  deleteLink: (id: string) => Promise<void>;
  reorderLinks: (linkIds: string[]) => Promise<void>;
}

export const LinksProvider = ({ children, initialLinks }: LinksProviderProps) => {
  const [links, setLinks] = useState<Link[]>(initialLinks);

  const addLink = async (data: CreateLinkInput) => {
    // 乐观更新
    const tempId = `temp-${Date.now()}`;
    const optimisticLink: Link = {
      id: tempId,
      ...data,
      createdAt: new Date(),
      updatedAt: new Date(),
    };
    setLinks((prev) => [...prev, optimisticLink]);

    try {
      const result = await createLink(data);
      if (!result.success) throw new Error(result.error);

      setLinks((prev) =>
        prev.map((link) => (link.id === tempId ? result.data : link))
      );
    } catch (error) {
      setLinks((prev) => prev.filter((link) => link.id !== tempId));
      toast.error(error.message);
    }
  };

  // 其他方法...

  return (
    <LinksContext.Provider value={{ links, addLink, updateLink, deleteLink, reorderLinks }}>
      {children}
    </LinksContext.Provider>
  );
};
```

### 2. 使用 SWR 管理服务器状态

```typescript
// src/hooks/use-links.ts
import useSWR, { mutate } from 'swr';
import { getUserLinks as fetchUserLinks } from '@/actions/link-actions';

export function useLinks() {
  const { data, error, isLoading, mutate } = useSWR(
    'user-links',
    fetchUserLinks,
    {
      revalidateOnFocus: false,
      revalidateOnReconnect: false,
    }
  );

  const addLink = async (data: CreateLinkInput) => {
    // 乐观更新
    mutate(
      (current) => ({
        success: true,
        data: [...(current?.data ?? []), { ...data, id: `temp-${Date.now()}` }],
      }),
      false
    );

    // 实际更新
    const result = await createLink(data);
    if (result.success) {
      // 用服务器数据替换
      mutate();
    } else {
      // 回滚
      mutate();
      throw new Error(result.error);
    }
  };

  return {
    links: data?.data ?? [],
    isLoading,
    error,
    addLink,
    // ...
  };
}
```
